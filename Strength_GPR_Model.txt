import numpy as np
import matplotlib.pyplot as plt
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import ConstantKernel, RBF, DotProduct, WhiteKernel
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import r2_score, mean_absolute_error, mean_squared_error
import shap

print("Libraries loaded. Initializing Collinearity-Corrected Kernel...")

# 1. LOAD THE DATASET
# Features: [Cr, Ni, Al, Cu, AnnTemp] -> Target: [Yield Strength]
literature_data = np.array([
    [25.0, 25.0, 0.0, 0.0, 25, 180], [24.5, 24.5, 0.0, 0.0, 25, 400],
    [24.0, 24.0, 0.0, 0.0, 25, 450], [23.5, 23.5, 0.0, 0.0, 25, 350],
    [20.0, 40.0, 2.0, 0.0, 25, 220], [19.0, 38.0, 5.6, 0.0, 25, 220],
    [18.0, 36.0, 9.0, 0.0, 25, 300], [18.0, 36.0, 10.7, 0.0, 25, 400],
    [17.5, 35.0, 12.3, 0.0, 25, 600], [17.2, 34.5, 13.8, 0.0, 25, 750],
    [17.0, 34.0, 15.0, 0.0, 25, 1150], [16.6, 33.3, 16.6, 0.0, 25, 730],
    [40.0, 40.0, 0.0, 0.0, 850, 1091], [40.0, 40.0, 0.0, 0.0, 900, 1058],
    [40.0, 40.0, 0.0, 0.0, 950, 1009], [40.0, 40.0, 0.0, 0.0, 1000, 604],
    [40.0, 40.0, 0.0, 0.0, 1050, 584]
])

# CORRECTED COMPOSITION: Fe60Cr17Ni15-xAl8Cux
# FORMAT: [Cr, Ni, Al, Cu, AnnTemp, Yield Strength]
experimental_data = np.array([
    [17.0, 15.0, 8.0, 0.0, 900, 890],  # Cu0-900
    [17.0, 14.5, 8.0, 0.5, 900, 900],  # Cu0.5-900
    [17.0, 14.0, 8.0, 1.0, 900, 950],  # Cu1-900
    [17.0, 15.0, 8.0, 0.0, 1000, 690], # Cu0-1000
    [17.0, 14.5, 8.0, 0.5, 1000, 730], # Cu0.5-1000
    [17.0, 14.0, 8.0, 1.0, 1000, 800]  # Cu1-1000
])

feature_names = ['Cr', 'Ni', 'Al', 'Cu', 'AnnTemp']

# UPGRADE: DotProduct + RBF Kernel to handle the Ni/Cu Collinearity
kernel = ConstantKernel(1.0) * RBF(length_scale=np.ones(5)) + DotProduct(sigma_0=1.0) + WhiteKernel(noise_level=0.1)

print("\nExecuting LOOCV with Tikhonov Regularization...")
y_pred_loocv = []

for i in range(len(experimental_data)):
    X_test_fold = experimental_data[i:i+1, :5]
    train_exp = np.delete(experimental_data, i, axis=0)
    
    # Clean, targeted augmentation to define the physical planes
    X_aug, y_aug = [], []
    for j in range(len(literature_data)):
        X_aug.append(literature_data[j, :5])
        y_aug.append(literature_data[j, 5])
            
    # Anchor the experimental data
    for j in range(len(train_exp)):
        X_aug.append(train_exp[j, :5])
        y_aug.append(train_exp[j, 5])
        for _ in range(10): # Reduced loop because the collinearity is fixed
            X_aug.append(train_exp[j, :5] * (1 + np.random.normal(0, 0.01, 5)))
            y_aug.append(train_exp[j, 5]) 
            
    X_train_fold = np.array(X_aug)
    y_train_fold = np.array(y_aug)
    
    scaler = StandardScaler()
    X_train_scaled = scaler.fit_transform(X_train_fold)
    X_test_scaled = scaler.transform(X_test_fold)
    
    # CRITICAL FIX: alpha=0.5 prevents the Ni+Cu=15.0 matrix collapse
    gpr = GaussianProcessRegressor(kernel=kernel, alpha=0.5, n_restarts_optimizer=20, normalize_y=True, random_state=42)
    gpr.fit(X_train_scaled, y_train_fold)
    y_pred_loocv.append(gpr.predict(X_test_scaled)[0])

y_test_true = experimental_data[:, 5]
r2 = r2_score(y_test_true, y_pred_loocv)
mae = mean_absolute_error(y_test_true, y_pred_loocv)
rmse = np.sqrt(mean_squared_error(y_test_true, y_pred_loocv))

print(f"LOOCV Complete! Parity R2 = {r2:.3f}")

# ---------------------------------------------------------
# PLOT E: PARITY PLOT FOR YIELD STRENGTH
# ---------------------------------------------------------
plt.figure(figsize=(6, 5))
plt.scatter(y_test_true, y_pred_loocv, color='royalblue', alpha=0.8, edgecolors='k', s=60, zorder=3)

padding = (max(y_test_true) - min(y_test_true)) * 0.2
min_val = min(min(y_test_true), min(y_pred_loocv)) - padding
max_val = max(max(y_test_true), max(y_pred_loocv)) + padding
plt.plot([min_val, max_val], [min_val, max_val], color='blue', linestyle='-', linewidth=1.5)

z = np.polyfit(y_test_true, y_pred_loocv, 1)
p = np.poly1d(z)
plt.plot(y_test_true, p(y_test_true), "g--", linewidth=2)

textstr = f'$R^2$ = {r2:.3f}\nMAE = {mae:.1f}\nRMSE = {rmse:.1f}'
props = dict(boxstyle='round', facecolor='white', alpha=0.9, edgecolor='gray')
plt.gca().text(0.05, 0.95, textstr, transform=plt.gca().transAxes, fontsize=11, verticalalignment='top', bbox=props)

plt.xlabel('Experimental Yield Strength (MPa)', fontsize=12)
plt.ylabel('ML Predicted Yield Strength (MPa)', fontsize=12)
plt.title('(e) GPR Parity Plot for Yield Strength', fontsize=14, loc='left')
plt.grid(True, linestyle='--', alpha=0.5, zorder=0)
plt.tight_layout()
plt.savefig('Panel_E_YS_Parity.png', dpi=300)
plt.show()

# ---------------------------------------------------------
# PLOT F: SHAP SUMMARY PLOT
# ---------------------------------------------------------
print("Calculating SHAP values for Yield Strength...")
X_aug_master, y_aug_master = [], []

for j in range(len(literature_data)):
    X_aug_master.append(literature_data[j, :5])
    y_aug_master.append(literature_data[j, 5])
        
for j in range(len(experimental_data)):
    X_aug_master.append(experimental_data[j, :5])
    y_aug_master.append(experimental_data[j, 5])
    for _ in range(10):
        X_aug_master.append(experimental_data[j, :5] * (1 + np.random.normal(0, 0.01, 5)))
        y_aug_master.append(experimental_data[j, 5])
        
X_train_master = np.array(X_aug_master)
y_train_master = np.array(y_aug_master)

scaler_master = StandardScaler()
X_master_scaled = scaler_master.fit_transform(X_train_master)
X_exp_scaled = scaler_master.transform(experimental_data[:, :5])

gpr_master = GaussianProcessRegressor(kernel=kernel, alpha=0.5, n_restarts_optimizer=20, normalize_y=True, random_state=42)
gpr_master.fit(X_master_scaled, y_train_master)

explainer = shap.KernelExplainer(gpr_master.predict, shap.kmeans(X_master_scaled, 10))
shap_values = explainer.shap_values(X_exp_scaled)

plt.figure(figsize=(7, 4))
shap.summary_plot(shap_values, X_exp_scaled, feature_names=feature_names, show=False)
plt.title('(f) SHAP Summary Plot for Yield Strength', fontsize=14, loc='left')
plt.tight_layout()
plt.savefig('Panel_F_YS_SHAP.png', dpi=300)
plt.show()

print("SUCCESS! Panels E and F are ready for download.")
