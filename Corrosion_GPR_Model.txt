import numpy as np
import matplotlib.pyplot as plt
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import ConstantKernel, Matern, WhiteKernel
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import r2_score, mean_absolute_error, mean_squared_error
import shap

print("Libraries loaded. Starting Target-Pinned Analysis...")

# 1. LOAD THE UNIFIED DATASET
literature_data = np.array([
    [11.1, 11.1, 5.5, 0.0, 300, -0.266, 0.424e-6], [11.1, 11.1, 5.5, 0.0, 375, -0.213, 0.130e-6],
    [11.1, 11.1, 5.5, 0.0, 450, -0.233, 0.310e-6], [11.1, 11.1, 5.5, 0.0, 525, -0.229, 0.470e-6],
    [11.1, 11.1, 5.5, 0.0, 600, -0.296, 0.374e-6], [11.1, 11.1, 5.5, 0.0, 800, -0.237, 1.400e-6],
    [11.1, 11.1, 5.5, 0.0, 900, -0.123, 3.700e-6], [11.9, 10.0, 0.0, 0.0, 1200, -0.860, 4.310e-5],
    [5.94, 5.26, 0.0, 0.0, 1200, -0.720, 3.410e-5], [11.9, 10.0, 0.0, 0.0, 1300, -0.930, 3.910e-5],
    [5.94, 5.26, 0.0, 0.0, 1300, -0.810, 3.100e-5], [25.0, 25.0, 0.0, 0.0, 25, -1.024, 1.660e-5],
    [24.5, 24.5, 0.0, 0.0, 25, -1.020, 1.470e-5], [24.0, 24.0, 0.0, 0.0, 25, -1.030, 2.570e-5],
    [18.1, 27.2, 0.0, 0.0, 1100, -0.920, 0.300e-7], [17.8, 26.7, 0.0, 0.0, 1100, -0.710, 0.780e-7],
    [16.6, 25.0, 0.0, 0.0, 1100, -0.640, 0.720e-7], [19.2, 46.1, 11.5, 0.0, 25, -0.159, 2.600e-7]
])

experimental_data = np.array([
    [17.5, 13.0, 8.0, 0.0, 900, -0.369, 2.94e-5], [17.5, 12.5, 8.0, 0.5, 900, -0.301, 3.95e-6],
    [17.5, 12.0, 8.0, 1.0, 900, -0.284, 4.99e-6], [17.5, 13.0, 8.0, 0.0, 1000, -0.364, 1.94e-5],
    [17.5, 12.5, 8.0, 0.5, 1000, -0.301, 2.89e-6], [17.5, 12.0, 8.0, 1.0, 1000, -0.279, 3.66e-6]
])

# Convert Icorr to log10
literature_data[:, 6] = np.log10(literature_data[:, 6])
experimental_data[:, 6] = np.log10(experimental_data[:, 6])

feature_names = ['Cr', 'Ni', 'Al', 'Cu', 'AnnTemp']

# Optimized Matern Kernel for high precision
kernel = ConstantKernel(1.0, (1e-3, 1e3)) * Matern(nu=2.5, length_scale=1.0) + WhiteKernel(noise_level=1e-3)

def run_ml_pipeline(target_col, ylabel, plot_a_name, plot_b_name, filename_prefix):
    print(f"\n---> Starting Pinned Analysis for {plot_a_name} <---")
    y_pred_loocv = []
    
    for i in range(len(experimental_data)):
        X_test_fold = experimental_data[i:i+1, :5]
        train_exp = np.delete(experimental_data, i, axis=0)
        
        X_aug, y_aug = [], []
        
        # 1x Background Literature Data (with full noise)
        for j in range(len(literature_data)):
            X_aug.append(literature_data[j, :5])
            y_aug.append(literature_data[j, target_col])
            X_aug.append(literature_data[j, :5] * (1 + np.random.normal(0, 0.01, 5)))
            y_aug.append(literature_data[j, target_col] + np.random.normal(0, 0.01))
                
        # 30x Target-Pinned Experimental Data
        for j in range(len(train_exp)):
            X_aug.append(train_exp[j, :5])
            y_aug.append(train_exp[j, target_col])
            for _ in range(30):
                # Noise on features (X), NO NOISE on target (Y)
                X_aug.append(train_exp[j, :5] * (1 + np.random.normal(0, 0.01, 5)))
                y_aug.append(train_exp[j, target_col]) 
                
        X_train_fold = np.array(X_aug)
        y_train_fold = np.array(y_aug)
        
        scaler = StandardScaler()
        X_train_scaled = scaler.fit_transform(X_train_fold)
        X_test_scaled = scaler.transform(X_test_fold)
        
        gpr = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=20, normalize_y=True, random_state=42)
        gpr.fit(X_train_scaled, y_train_fold)
        y_pred_loocv.append(gpr.predict(X_test_scaled)[0])

    y_test_true = experimental_data[:, target_col]
    r2 = r2_score(y_test_true, y_pred_loocv)
    mae = mean_absolute_error(y_test_true, y_pred_loocv)
    rmse = np.sqrt(mean_squared_error(y_test_true, y_pred_loocv))

    print(f"LOOCV Complete! Parity R2 = {r2:.3f}")
    
    plt.figure(figsize=(6, 5))
    plt.scatter(y_test_true, y_pred_loocv, color='royalblue', alpha=0.8, edgecolors='k', s=60, zorder=3)
    
    padding = (max(y_test_true) - min(y_test_true)) * 0.2
    min_val = min(min(y_test_true), min(y_pred_loocv)) - padding
    max_val = max(max(y_test_true), max(y_pred_loocv)) + padding
    plt.plot([min_val, max_val], [min_val, max_val], color='blue', linestyle='-', linewidth=1.5)
    
    z = np.polyfit(y_test_true, y_pred_loocv, 1)
    p = np.poly1d(z)
    plt.plot(y_test_true, p(y_test_true), "g--", linewidth=2)
    
    textstr = f'$R^2$ = {r2:.3f}\nMAE = {mae:.3f}\nRMSE = {rmse:.3f}'
    props = dict(boxstyle='round', facecolor='white', alpha=0.9, edgecolor='gray')
    plt.gca().text(0.05, 0.95, textstr, transform=plt.gca().transAxes, fontsize=11, verticalalignment='top', bbox=props)
    
    plt.xlabel(f'Experimental {ylabel}', fontsize=12)
    plt.ylabel(f'ML Predicted {ylabel}', fontsize=12)
    plt.title(plot_a_name, fontsize=14, loc='left')
    plt.grid(True, linestyle='--', alpha=0.5, zorder=0)
    plt.tight_layout()
    plt.savefig(f'{filename_prefix}_Parity.png', dpi=300)
    plt.show()

    print("Calculating SHAP values...")
    X_aug_master, y_aug_master = [], []
    for j in range(len(literature_data)):
        X_aug_master.append(literature_data[j, :5])
        y_aug_master.append(literature_data[j, target_col])
        X_aug_master.append(literature_data[j, :5] * (1 + np.random.normal(0, 0.01, 5)))
        y_aug_master.append(literature_data[j, target_col] + np.random.normal(0, 0.01))
            
    for j in range(len(experimental_data)):
        X_aug_master.append(experimental_data[j, :5])
        y_aug_master.append(experimental_data[j, target_col])
        for _ in range(30):
            X_aug_master.append(experimental_data[j, :5] * (1 + np.random.normal(0, 0.01, 5)))
            y_aug_master.append(experimental_data[j, target_col])
            
    X_train_master = np.array(X_aug_master)
    y_train_master = np.array(y_aug_master)
    
    scaler_master = StandardScaler()
    X_master_scaled = scaler_master.fit_transform(X_train_master)
    X_exp_scaled = scaler_master.transform(experimental_data[:, :5])
    
    gpr_master = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=20, normalize_y=True, random_state=42)
    gpr_master.fit(X_master_scaled, y_train_master)
    
    explainer = shap.KernelExplainer(gpr_master.predict, shap.kmeans(X_master_scaled, 10))
    shap_values = explainer.shap_values(X_exp_scaled)
    
    plt.figure(figsize=(7, 4))
    shap.summary_plot(shap_values, X_exp_scaled, feature_names=feature_names, show=False)
    plt.title(plot_b_name, fontsize=14, loc='left')
    plt.tight_layout()
    plt.savefig(f'{filename_prefix}_SHAP.png', dpi=300)
    plt.show()

# Run for Ecorr
run_ml_pipeline(5, "$E_{corr}$ (V vs SCE)", "(a) GPR Parity Plot for $E_{corr}$", "(b) SHAP Summary Plot for $E_{corr}$", "Panel_A_B_Ecorr")

# Run for Icorr
run_ml_pipeline(6, "$\log(I_{corr})$ (A/cm²)", "(c) GPR Parity Plot for $I_{corr}$", "(d) SHAP Summary Plot for $I_{corr}$", "Panel_C_D_Icorr")
